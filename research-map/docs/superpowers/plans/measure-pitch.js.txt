// Layout check for deck 06, by measuring (screenshots of the decks time out in Claude in Chrome here).
// Paste in the console, or run through javascript_tool, on the deck's page in the ready state (every slide complete).
// Reports, per slide: elements outside the slide, clipped text, a title over a diagram (.pitch__fig),
// sources or content over the chapter bar. Run it in both themes and, if you like, in a narrower window.
(async () => {
  document.startViewTransition = undefined // slides swap at once while measuring
  const slides = [...document.querySelectorAll('.stage section.slide')]
  const R = (e) => e.getBoundingClientRect()
  const hit = (a, b) => a.left < b.right && a.right > b.left && a.top < b.bottom && a.bottom > b.top
  const found = []
  for (let i = 0; i < slides.length; i++) {
    location.hash = '#' + (i + 1)
    await new Promise((r) => setTimeout(r, 200))
    const s = slides[i]
    const sr = R(s)
    const bar = R(document.querySelector('.pitch__bar'))
    const issues = []
    for (const e of s.querySelectorAll('*')) {
      const r = R(e)
      if (!r.width || !r.height) continue
      if (r.left < sr.left - 1 || r.right > sr.right + 1 || r.top < sr.top - 1 || r.bottom > sr.bottom + 1)
        issues.push(`outside the slide: ${e.tagName}.${e.className}`)
      if (e.scrollWidth > e.clientWidth + 1 && getComputedStyle(e).overflowX !== 'visible')
        issues.push(`clipped text: ${e.tagName}.${e.className}`)
    }
    const h1 = s.querySelector('h1')
    if (h1) for (const f of s.querySelectorAll('.pitch__fig')) if (hit(R(h1), R(f))) issues.push('title over a figure')
    const refs = s.querySelector('.refs')
    if (refs && hit(R(refs), bar)) issues.push('sources over the bar')
    for (const e of s.querySelectorAll('h1, p:not(.refs), .pitch__fig, .pitch__tag'))
      if (hit(R(e), bar)) issues.push(`over the bar: ${e.tagName}.${e.className}`)
    if (issues.length) found.push(`${i + 1}: ${[...new Set(issues)].join(' | ')}`)
  }
  return found.length ? found : 'no findings'
})()
